p0 观察者模式：最小可用版

function Promise(exector) {
  this.pending = [];        // 观察者数组
  this.value = undefined;
  const resolve = value => {
    if (this.pending) {
      this.value = value;
      for (const onFulfilled of this.pending) {
        onFulfilled(this.value);   // 通知观察者
      }
      this.pending = undefined;
    }
  };
  exector(resolve);
}

p2-p3 状态传递：thenable 统一决议值

function isPromise(value) {
  return value && typeof value.then === 'function';  // 鸭子类型
}
function wrapToThenable(value) {
  if (isPromise(value)) return value;   // Promise 原样返回
  return { then: onFulfilled => wrapToThenable(onFulfilled(value)) };
}
function wrapToRejected(value) {
  return { then: (_, onRejected) => wrapToThenable(onRejected(value)) };
}
决议值 包装结果 then 的行为
普通值 wrapToThenable 包装 立即调用 onFulfilled
已决议 Promise 原样返回 立即执行回调
未决议 Promise 原样返回 注册回调，等它决议
reject 的失败值 wrapToRejected 包装 只调用 onRejected
关键设计

三种决议值都变成"有 then 的对象"，通知逻辑不需要分支。value 是未决议的 Promise 时，通知责任转移到它身上，这就是"通知转移"。

p4 异常处理：两处捕获 + catch

function returnWhenError(onError) {
  return fun => (...args) => {
    let result;
    try {
      result = fun(...args);
    } catch (e) {
      result = onError(e);      // 异常转为失败值
    }
    return result;
  };
}
Promise.prototype.catch = function (onRejected) {
  return Promise.prototype.then.call(this, undefined, onRejected);
};
